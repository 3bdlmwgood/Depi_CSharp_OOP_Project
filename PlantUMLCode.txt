' =========================
' Interfaces
' =========================

interface ICloneable {
    +Clone() : object
}

interface "IComparable<T>" as IComparable {
    +CompareTo(T other) : int
}


' =========================
' Answer
' =========================

class Answer {
    +int Id
    +string Text

    +Answer()
    +Answer(int answerId, string answerText)
    +ToString() : string
}


' =========================
' Questions
' =========================

abstract class Question {
    +string Header
    +string Body
    +double Mark
    +List<Answer> AnswerList
    +Answer RightAnswer

    +Question()
    +Question(...)
    +Clone() : object
    +CompareTo(Question other) : int
    +ToString() : string
}

class MCQQuestion {
    +MCQQuestion()
    +MCQQuestion(...)
    +ToString() : string
}

class TrueFalseQuestion {
    +TrueFalseQuestion()
    +TrueFalseQuestion(...)
    +ToString() : string
}


' =========================
' Exam
' =========================

enum enExamType {
    Final
    Practical
}

abstract class Exam {
    +int TimeOfExam
    +int NumberOfQuestions
    +enExamType ExamType
    +List<Question> Questions

    +Exam()
    +Exam(...)
    +ShowExam() : void
    +Clone() : object
    +CompareTo(Exam other) : int
    +ToString() : string
}

class FinalExam {
    +FinalExam()
    +FinalExam(...)
    +ShowExam() : void
    +ToString() : string
}

class PracticalExam {
    +PracticalExam()
    +PracticalExam(...)
    +ShowExam() : void
    +ToString() : string
}


' =========================
' Subject
' =========================

class Subject {
    +int SubjectId
    +string SubjectName
    +Exam Exam

    +Subject()
    +Subject(int subjectId, string subjectName)
    +CreateExam(...) : Exam
    +ToString() : string
}
